    <script>
        // Initialize wallet balances
        let profitWallet = 0.00;
        let rechargeWallet = 0.00;

        // Load wallet balances from local storage
        if (localStorage.getItem('profitWallet')) {
            profitWallet = parseFloat(localStorage.getItem('profitWallet'));
        }
        if (localStorage.getItem('rechargeWallet')) {
            rechargeWallet = parseFloat(localStorage.getItem('rechargeWallet'));
        }

        // Update wallet balances on page load
        document.getElementById('profit-wallet').textContent = `R${profitWallet.toFixed(2)}`;
        document.getElementById('recharge-wallet').textContent = `R${rechargeWallet.toFixed(2)}`;

        // Add event listener to recharge link
        document.getElementById('recharge-link').addEventListener('click', () => {
            // Create a new container for the recharge details
            const rechargeDetails = document.createElement('div');
            rechargeDetails.classList.add('recharge-container');

            // Generate a random reference number
            const referenceNumber = Math.floor(1000 + Math.random() * 9000);

            // Set the content of the recharge details
            rechargeDetails.innerHTML = `
                <h3>Recharge Details</h3>
                <p><strong>Own Amount</strong></p>
                <p><strong>Transfer Type:</strong> Bank</p>
                <p><strong>Bank Name:</strong> Tymebank</p>
                <p><strong>Account No:</strong> 51104524563</p>
                <p><strong>Reference:</strong> ${referenceNumber}</p>
                <p><strong>Enter Amount:</strong></p>
                <input type="number" id="recharge-amount" placeholder="Enter amount">
                <button id="confirm-payment">Confirm Payment</button>
            `;

            // Append the recharge details to the main content
            const mainContent = document.querySelector('.main-content');
            mainContent.innerHTML = '';  // Clear current content
            mainContent.appendChild(rechargeDetails);

            // Add event listener to the Confirm Payment button
            document.getElementById('confirm-payment').addEventListener('click', () => {
                const rechargeAmount = parseFloat(document.getElementById('recharge-amount').value);
                if (!isNaN(rechargeAmount) && rechargeAmount > 0) {
                    rechargeWallet += rechargeAmount;
                    localStorage.setItem('rechargeWallet', rechargeWallet.toString());
                    document.getElementById('recharge-wallet').textContent = `R${rechargeWallet.toFixed(2)}`;
                    alert('Recharge successful!');
                } else {
                    alert('Please enter a valid amount.');
                }
            });
        });

        // Add event listener to the "Home" link
        document.getElementById('home-link').addEventListener('click', () => {
            location.reload();  // Reload the page to go back to the main menu
        });

        // Add event listener to the "Group" link
        document.getElementById('group-link').addEventListener('click', () => {
            window.location.href = "https://chat.whatsapp.com/LZeDpHpdqa00mnxpCBWUyO";  // Redirect to WhatsApp group
        });
    </script>
